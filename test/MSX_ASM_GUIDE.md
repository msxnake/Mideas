## ⚠️ **ESTRUCTURA DE MEMORIA MSX - CRÍTICA**

### **📍 Organización de memoria correcta:**

```
MSX Memory Layout:
#0000 - #BFFF : ROM (solo lectura)  ← Código, datos constantes, assets
#C000 - #FFFF : RAM (escribible)   ← Variables proyecto: #C000-#F37F, sistema: #F380+

ROM Cartucho: #4000 - #BFFF        ← Nuestro código va aquí
Variables:    #C000-#F37F           ← Variables del proyecto (NO tocar #F380+)
```

### **🚨 ERRORES COMUNES:**
❌ **NO** poner variables en zona ROM (#4000-#BFFF) - son solo lectura
❌ **NO** limpiar zona del sistema (#F380-#FFFF) - causa reseteo/crash
✅ **SÍ** poner variables en zona RAM proyecto (#C000-#F37F) - escribibles y seguras

### **🎯 Estructura correcta del ASM:**
1. `ORG #4000` - Inicio del código ROM
2. Cabecera ROM, código, funciones, datos constantes
3. Variables con `EQU` (direcciones fijas #C000-#F37F)
4. Funciones de inicialización segura de RAM del proyecto
5. **NO usar ORG #C000 + DB/DS** - usar EQU para evitar conflictos

### **3.1 CABECERA Y COMENTARIOS**
```assembly
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
;; NOMBRE_PROYECTO - UNITED FILES ASM (ALL-IN-ONE)
;; Generated by MSX Retro Game IDE
;; Cartridge ROM Format
;; Target: MSX1/MSX2/MSX2+
;;
;; MEMORY LAYOUT MSX:
;; #0000-#BFFF: ROM del sistema (solo lectura)
;; #C000-#FFFF: RAM (lectura/escritura - variables aquí)
;;
;; Base Address: #4000 (ROM cartucho)
;; Variables RAM: #C000+ (zona escribible)
;; Generated: [TIMESTAMP]
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

; ==================================================================
; MEMORY ORGANIZATION (DEBE SER LO PRIMERO)
; ==================================================================
    ORG #4000                ; ROM cartucho - OBLIGATORIO AL INICIO
```

### **3.2 CABECERA ROM CARTUCHO**
```assembly
; ==================================================================
; CARTRIDGE ROM HEADER
; ==================================================================
    DB "AB"              ; ROM signature (OBLIGATORIO)
    DW INIT_ROM          ; Init address (punto de entrada)
    DW 0,0,0,0,0         ; Statement, Device, Text, Reserved

INIT_ROM:
    ; Cartridge initialization
    DI                   ; Disable interrupts during init

    ; Initialize stack for cartridge
    LD SP, #F380

    ; ⚠️ CRÍTICO: Inicializar solo variables del proyecto (NO del sistema)
    CALL INIT_RAM_VARIABLES     ; Solo variables #C000-#C520

    ; Initialize VDP for games
    CALL INIT_VDP_ROM

    ; Initialize sound (if needed)
    CALL INIT_SOUND_ROM

    ; Verificar límites de RAM (debug)
    CALL VERIFY_RAM_BOUNDS

    ; Jump to main game
    JP MAIN_GAME_START
```

### **5.0 ANÁLISIS DEL PROYECTO - Detección de Assets**
```typescript
// PRIMERO: Analizar qué assets existen realmente en el proyecto
function analyzeProjectAssets(projectData) {
    const analysis = {
        hasGameFlow: false,
        screens: [],
        sprites: [],
        tiles: [],
        fonts: [],
        worlds: [],
        stateMachines: [],
        hasCollisions: false,
        hasSound: false,
        hasInput: false
    };

    // Verificar Game Flow (OBLIGATORIO)
    if (projectData.gameFlow && projectData.gameFlow.states.length > 0) {
        analysis.hasGameFlow = true;
        analysis.gameFlow = projectData.gameFlow;
    } else {
        throw new Error("❌ ERROR: No Game Flow encontrado - No se puede generar ROM");
    }

    // Analizar cada editor
    if (projectData.screens && projectData.screens.length > 0) {
        analysis.screens = projectData.screens.filter(s => s.data && s.data.length > 0);
    }

    if (projectData.sprites && projectData.sprites.length > 0) {
        analysis.sprites = projectData.sprites.filter(s => s.patternData && s.patternData.length > 0);
    }

    if (projectData.tiles && projectData.tiles.length > 0) {
        analysis.tiles = projectData.tiles.filter(t => t.patternData && t.patternData.length > 0);
    }

    if (projectData.fonts && projectData.fonts.length > 0) {
        analysis.fonts = projectData.fonts.filter(f => f.characters && f.characters.length > 0);
    }

    if (projectData.worlds && projectData.worlds.length > 0) {
        analysis.worlds = projectData.worlds.filter(w => w.mapData && w.mapData.length > 0);
    }

    if (projectData.stateMachines && projectData.stateMachines.length > 0) {
        analysis.stateMachines = projectData.stateMachines.filter(sm => sm.states && sm.states.length > 0);
    }

    // Detectar sistemas necesarios
    analysis.hasCollisions = analysis.sprites.length > 0 || analysis.worlds.some(w => w.entities);
    analysis.hasInput = analysis.gameFlow.states.some(s => s.transitions?.some(t => t.trigger === 'input'));

    return analysis;
}
```

import { generateScreenLayoutASMCode } from './components/utils/screenUtils';

function generateScreenData(screens) {
    if (screens.length === 0) return '';

    let code = `; ==================================================================\n`;
    code += `; SCREEN DATA FROM SCREEN EDITOR (usando función probada)\n`;
    code += `; ==================================================================\n\n`;

    screens.forEach((screen, screenIndex) => {
        // Usar la función existente que ya funciona correctamente
        const screenASM = generateScreenLayoutASMCode(
            screen.name,
            screen.width,
            screen.height,
            screen.mapIndices, // Array de índices de tiles ya procesados
            screen.referenceComments || [], // Comentarios opcionales
            'hex' // Formato hexadecimal
        );

        code += screenASM + '\n';
    });

    // Tabla de screens para navegación (si hay múltiples)
    if (screens.length > 1) {
        code += `SCREEN_COUNT    EQU ${screens.length}\n`;
        code += `SCREEN_TABLE:\n`;
        screens.forEach((screen, index) => {
            const safeMapName = screen.name.replace(/[^a-zA-Z0-9_]/g, '_').toUpperCase();
            code += `    DW SCREEN_${safeMapName}_LAYOUT\n`;
        });
        code += `\n`;
    }

    return code;
}

### **Sistema de Renderizado - Copia Screen1 → Screen2**

```assembly
; ==================================================================
; SCREEN COPYING SYSTEM - Screen1 to Screen2 Mode
; ==================================================================

COPY_SCREEN1_TO_SCREEN2:
    ; Copia contenido de Screen1 (texto) a Screen2 (gráficos)
    ; Usado para mostrar menús de texto en modo gráfico

    ; Set VRAM read address (Screen1 name table)
    LD HL, #1800            ; Screen1 name table
    LD DE, #1800            ; Screen2 name table
    LD BC, 768              ; 32x24 = 768 bytes

COPY_SCREEN_LOOP:
    ; Read from Screen1 VRAM
    LD A, L
    OUT (VDPSR), A
    LD A, H
    OUT (VDPSR), A
    IN A, (VDPDR)           ; Read character code

    PUSH AF

    ; Write to Screen2 VRAM
    LD A, E
    OUT (VDPSR), A
    LD A, D
    OR #40                  ; Set write bit
    OUT (VDPSR), A

    POP AF
    OUT (VDPDR), A          ; Write character

    INC HL
    INC DE
    DEC BC
    LD A, B
    OR C
    JR NZ, COPY_SCREEN_LOOP

    RET

DRAW_SCREEN_TO_SCREEN2:
    ; Dibuja una pantalla diseñada en Screen Editor a Screen2
    ; A = Screen ID to draw

    ; Calcular puntero a datos de pantalla
    LD L, A
    LD H, 0
    ADD HL, HL
    LD DE, SCREEN_TABLE
    ADD HL, DE
    LD E, (HL)
    INC HL
    LD D, (HL)              ; DE = pointer to screen data

    ; Configurar destino en VRAM
    LD HL, #1800            ; Screen2 name table base

    ; Obtener dimensiones de pantalla
    LD A, (DE)              ; Width
    LD B, A
    INC DE
    LD A, (DE)              ; Height
    LD C, A
    INC DE

DRAW_SCREEN_ROW:
    PUSH BC
    LD B, A                 ; B = width

DRAW_SCREEN_COL:
    ; Leer tile ID de screen data
    LD A, (DE)
    INC DE

    ; Escribir a VRAM
    PUSH HL
    PUSH DE
    EX DE, HL
    LD (temp_byte), A
    LD HL, temp_byte
    CALL VRAM_WRITE_BYTE
    POP DE
    POP HL

    INC HL
    DEC B
    JR NZ, DRAW_SCREEN_COL

    POP BC
    DEC C
    JR NZ, DRAW_SCREEN_ROW

    RET

; Variables para sistema de pantalla
current_screen_id:      DB 0
temp_byte:              DB 0
```

### **🔧 Funciones para gestionar variables RAM con EQU:**

```assembly
; ==================================================================
; INICIALIZACIÓN DE VARIABLES RAM (usando direcciones EQU)
; ==================================================================

INIT_RAM_VARIABLES:
    ; Inicializar todas las variables en RAM a valores por defecto
    ; ⚠️ CRÍTICO: Solo limpiar variables del proyecto, NO del sistema

    ; Método 1: Limpiar zona específica del proyecto (SEGURO)
    LD HL, current_flow_state   ; #C000 - primera variable del proyecto
    LD DE, current_flow_state+1 ; #C001 - siguiente byte
    LD BC, RAM_USAGE_END - current_flow_state - 1 ; Solo hasta #C520
    LD (HL), 0                  ; Poner primer byte a 0
    LDIR                        ; Limpia SOLO variables del proyecto

    ; ⚠️ IMPORTANTE: NO limpiar más allá de #F380 (variables del sistema MSX)

    ; Método 2: Inicializar variables individuales (más control)
    ; XOR A                     ; A = 0
    ; LD (current_flow_state), A
    ; LD (prev_flow_state), A
    ; LD (input_state), A
    ; LD (active_sprite_count), A
    ; ... etc

    ; Inicializar valores específicos no-cero
    LD A, 255
    LD (prev_flow_state), A     ; Estado anterior = no válido

    ; Inicializar punteros/words a 0
    LD HL, 0
    LD (flow_state_timer), HL
    LD (current_screen_data), HL
    LD (temp_word), HL
    LD (frame_counter), HL

    RET

VERIFY_RAM_BOUNDS:
    ; Verificar que las variables del proyecto no invadan zona del sistema
    ; ⚠️ DEBUG: Usar durante desarrollo

    ; Verificar que RAM_USAGE_END < #F380 (límite antes de variables sistema)
    LD HL, RAM_USAGE_END
    LD DE, #F380                ; Límite antes de variables del sistema MSX
    OR A                        ; Clear carry
    SBC HL, DE
    JR C, RAM_BOUNDS_OK         ; Si HL < #F380, OK

    ; ERROR: Variables del proyecto invaden zona del sistema MSX
    ; Esto corromperá variables críticas del BIOS/BASIC
    HALT                        ; Parar programa - error crítico

RAM_BOUNDS_OK:
    RET

; ==================================================================
; FUNCIONES AUXILIARES PARA TRABAJAR CON VARIABLES EQU
; ==================================================================

CLEAR_SPRITE_BUFFER:
    ; Limpiar buffer de sprites específicamente
    LD HL, sprite_attributes
    LD DE, sprite_attributes+1
    LD BC, 127                  ; 128 bytes - 1
    LD (HL), #D1               ; Y=209 (sprite invisible)
    LDIR
    RET

CLEAR_SCREEN_BUFFER:
    ; Limpiar buffer de pantalla específicamente
    LD HL, screen_buffer
    LD DE, screen_buffer+1
    LD BC, 767                  ; 768 bytes - 1
    LD (HL), 0                 ; Tile ID 0
    LDIR
    RET

CLEAR_WORK_BUFFER:
    ; Limpiar buffer de trabajo
    LD HL, work_buffer
    LD DE, work_buffer+1
    LD BC, 255                  ; 256 bytes - 1
    LD (HL), 0
    LDIR
    RET
```

    ; A = Screen ID to load
    ; Destroys: A, BC, DE, HL

    ; Get screen data pointer
    LD L, A
    LD H, 0

    LD C, A
LOAD_SCREEN_COL:

    INC HL
    INC DE
    DEC C
    JR NZ, LOAD_SCREEN_COL
    DEC B
    JR NZ, LOAD_SCREEN_ROW

    RET

VRAM_WRITE_BYTE:
    ; Write byte in (HL) to VRAM address DE
    ; Preserves BC, HL, DE
    PUSH BC
    PUSH HL
    PUSH DE

    ; Set VRAM write address
    LD A, E
    OUT (VDPSR), A
    LD A, D
    OR #40                  ; Set write bit
    OUT (VDPSR), A

    ; Write data
    LD A, (HL)
    OUT (VDPDR), A

    POP DE
    POP HL
    POP BC
    RET

; ==================================================================
; SPRITE SYSTEM IMPLEMENTATION
; ==================================================================

INIT_SPRITE_SYSTEM:
    ; Initialize sprite system
    ; Load sprite patterns to VRAM
    LD HL, SPRITE_PATTERN_TABLE
    LD DE, SPRPAT           ; Sprite pattern table in VRAM
    LD BC, SPRITE_COUNT * 8 ; Each sprite is 8 bytes
    CALL COPY_TO_VRAM

    ; Clear sprite attribute table
    LD HL, SPRATR
    LD DE, SPRATR + 1
    LD BC, 31
    LD (HL), #D1           ; Y=209 (invisible)
    LDIR

    ; Initialize sprite variables
    XOR A
    LD (active_sprite_count), A
    RET

ADD_SPRITE:
    ; Add sprite to screen
    ; A = sprite pattern ID, B = X pos, C = Y pos, D = color
    ; Returns: A = sprite slot (255 if full)

    PUSH BC
    PUSH DE

    ; Find free sprite slot
    LD A, (active_sprite_count)
    CP 32
    JR Z, ADD_SPRITE_FULL

    ; Calculate sprite attribute address
    LD L, A
    LD H, 0
    ADD HL, HL
    ADD HL, HL              ; HL = slot * 4
    LD DE, sprite_attributes
    ADD HL, DE              ; HL = attribute pointer

    ; Store sprite data
    LD (HL), C              ; Y position
    INC HL
    LD (HL), B              ; X position
    INC HL
    POP DE
    LD (HL), E              ; Pattern ID
    INC HL
    LD (HL), D              ; Color

    ; Increment active count
    LD A, (active_sprite_count)
    INC A
    LD (active_sprite_count), A
    DEC A                   ; Return slot number

    POP BC
    RET

ADD_SPRITE_FULL:
    POP DE
    POP BC
    LD A, 255               ; Error code
    RET

UPDATE_SPRITES:
    ; Update hardware sprite table from RAM buffer
    LD HL, sprite_attributes
    LD DE, SPRATR
    LD A, (active_sprite_count)
    OR A
    RET Z                   ; No sprites active

    LD B, A
UPDATE_SPRITE_LOOP:
    ; Copy 4 bytes per sprite
    CALL VRAM_WRITE_BYTE
    INC HL
    INC DE
    CALL VRAM_WRITE_BYTE
    INC HL
    INC DE
    CALL VRAM_WRITE_BYTE
    INC HL
    INC DE
    CALL VRAM_WRITE_BYTE
    INC HL
    INC DE

    DEC B
    JR NZ, UPDATE_SPRITE_LOOP
    RET

    ; DE = destination address in VRAM
